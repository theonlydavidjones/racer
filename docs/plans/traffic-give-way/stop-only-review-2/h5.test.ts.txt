    const tr = new Traffic(true, SEED), T0 = 21.6 * 3600;
    // warm
    const mk = (t: number) => { const vs = tr.vessels(t); const boats: Moving[] = []; const tg = near === 1 ?     let boats = mk(T0);
    const N = 60 * 600; let tot = 0, peak = 0, totBase = 0; const per: number[] = [];
    for (let rep = 0; rep < 2; rep++) { tot = 0; peak = 0; totBase = 0; per.length = 0;
      for (let k = 0; k < N; k++) { const t = T0 + k * DT; if (k % 600 === 0) boats = mk(t);
        const a = performance.now(); tr.vessels(t, boats, DT); const b = performance.now(); tr.vessels(t); const c = performance.now();
        tot += b - a; totBase += c - b; if (b - a > peak) peak = b - a; per.push(b - a); for (const q of boats) { q.x += q.vx * DT; q.y += q.vy * DT; } } }
    per.sort((x, y) => x - y);
    P(`${nb} boats near ${near} vessel(s): mean ${(tot / N * 1000).toFixed(1)} us a step (the timetable alone ${(totBase / N * 1000).toFixed(1)} us); p99 ${(per[Math.floor(N * 0.99)] * 1000).toFixed(0)} us, p99.9 ${(per[Math.floor(N * 0.999)] * 1000).toFixed(0)} us, peak ${(peak * 1000).toFixed(0)} us`);
