import { appendFileSync, writeFileSync } from 'node:fs';
import { it } from 'vitest';
import { SERVICES, Traffic, VESSEL, vesselHull, type Moving } from '/home/user/stop/src/sim/traffic';
import { segPt } from '/home/user/stop/src/core/math';
const OUT = '/tmp/claude-0/-home-user-racer/781bdda1-9a86-5cd6-bb3f-ed2540f513a9/scratchpad/rev/r5.txt';
writeFileSync(OUT, '');
const P = (...a: any[]) => appendFileSync(OUT, a.join(' ') + '\n');
const SEED = 2434542977, DT = 1 / 60, D2R = Math.PI / 180;
it('she passes a converging boat, slows, and calls it the overtaker', () => {
  for (const i of [3, 4, 2, 6]) {
    const s = SERVICES[i], tr0 = new Traffic(true, SEED), V = VESSEL[s.kind].kt * 0.5144;
    let T = 0; for (let t = 100; t < 40000; t += 1) { const v0 = tr0.vessels(t)[i], v1 = tr0.vessels(t + 200)[i]; if (v0.moving && v1.moving && Math.abs(v0.h - v1.h) < 0.01 && Math.abs(v0.h - tr0.vessels(t + 100)[i].h) < 0.01) { T = t; break; } }
    const v0 = tr0.vessels(T)[i], ux = Math.sin(v0.h * D2R), uy = Math.cos(v0.h * D2R);
    let n = 0, hits = 0, hitsTT = 0; const ex: string[] = [];
    for (const sp of [2.2, 2.7, 3.2]) for (const off of [10, 20, 35, 50]) for (const sgn of [1, -1]) for (const meetT of [40, 60, 80, 100]) for (const along of [-40, 0, 40, 80]) {
      // a boat converging on her route at `off` degrees, aimed to be `along` m ahead of where she'd be (on her timetable) at meetT
      const c = (v0.h + sgn * off) * D2R, bx = Math.sin(c) * sp, by = Math.cos(c) * sp;
      const mx = v0.x + ux * (V * meetT + along), my = v0.y + uy * (V * meetT + along), b0 = { x: mx - bx * meetT, y: my - by * meetT, vx: bx, vy: by };
      const go = (skip?: (b: Moving) => boolean) => { const tr = new Traffic(true, SEED), b = { ...b0 }; let least = Infinity, vAt = 0, tAt = 0, cosAt = 0, firstCos = NaN; const trc: string[] = []; let k = 0;
        for (let t = T; t < T + 190; t += DT) { const v = tr.vessels(t, [b], DT, skip)[i], hv = vesselHull(v), g = segPt(b, hv.a, hv.b) - hv.r, rx = b.x - v.x, ry = b.y - v.y, cs = (rx * ux + ry * uy) / Math.hypot(rx, ry);
          if (isNaN(firstCos)) firstCos = cs; if (g < least) { least = g; vAt = v.v; tAt = t - T; cosAt = cs; }
          if (k++ % 300 === 0) trc.push(`t${(t - T).toFixed(0)} k${tr.give[i].k.toFixed(2)} brg${(Math.acos(cs) / D2R).toFixed(0)} d${Math.hypot(rx, ry).toFixed(0)}${cs < -0.383 && bx * ux + by * uy > V * tr.give[i].k ? ' IGNORED(R13)' : ''}`);
          b.x += b.vx * DT; b.y += b.vy * DT; }
        return { least, vAt, tAt, cosAt, firstCos, trc }; };
      const a = go(), tt = go(() => true); n++;
      if (tt.least < 4) hitsTT++;
      if (a.least < 4 && a.vAt > 0.5) { hits++; if (ex.length < 2) ex.push(`   boat ${(sp / 0.5144).toFixed(1)} kt converging at ${off} deg, first seen ${(Math.acos(a.firstCos) / D2R).toFixed(0)} deg off her bow: contact at t ${a.tAt.toFixed(0)} s, she making ${(a.vAt / 0.5144).toFixed(1)} kt, the boat then ${(Math.acos(a.cosAt) / D2R).toFixed(0)} deg off her bow (timetable alone: ${tt.least.toFixed(0)} m)\n      ` + a.trc.join(' | ')); }
    }
    P(`${s.kind}#${i}: ${n} converging boats (slower than her along her route at the start, so she is the one coming up); contact while she makes way: ${hits} (timetable alone: ${hitsTT})`); for (const e of ex) P(e);
  }
});
