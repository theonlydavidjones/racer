const SEED = 2434542977, DT = 1 / 60, D2R = Math.PI / 180;
const gap = (b: Moving, v: any) => { const hv = vesselHull(v); return segPt(b, hv.a, hv.b) - hv.r; };
type Steer = (b: Moving, t: number) => void;
function run(i: number, T: number, boats0: Moving[], dur: number, opt: { skip?: (b: Moving) => boolean; steer?: Steer; log?: (t: number, tr: Traffic, v: any, bs: Moving[]) => void } = {}) {
  const tr = new Traffic(true, SEED), bs = boats0.map(b => ({ ...b }));
  let least = Infinity, minK = 1, vAt = 0;
  for (let t = T; t < T + dur; t += DT) {
    for (const b of bs) opt.steer?.(b, t - T);
    const v = tr.vessels(t, bs, DT, opt.skip)[i];
    for (const b of bs) { const g = gap(b, v); if (g < least) { least = g; vAt = v.v; } }
    minK = Math.min(minK, tr.give[i].k);
    opt.log?.(t - T, tr, v, bs);
    for (const b of bs) { b.x += b.vx * DT; b.y += b.vy * DT; }
  }
  return { least, minK, vAt, tr };
}
