const SEED = 2434542977, DT = 1 / 60, D2R = Math.PI / 180;
const gap = (b: Moving, v: any) => { const hv = vesselHull(v); return segPt(b, hv.a, hv.b) - hv.r; };
function run(i: number, T: number, b0: Moving, dur: number, skip?: (b: Moving) => boolean, trace?: string[]) {
  const tr = new Traffic(true, SEED), b = { ...b0 };
  let least = Infinity, vAt = 0, n = 0;
  for (let t = T; t < T + dur; t += DT) {
    const v = tr.vessels(t, [b], DT, skip)[i];
    const g = gap(b, v); if (g < least) { least = g; vAt = v.v; }
    if (trace && n++ % 600 === 0) { const ux = Math.sin(v.h * D2R), uy = Math.cos(v.h * D2R), rx = b.x - v.x, ry = b.y - v.y; trace.push(`t${(t - T).toFixed(0)} k${tr.give[i].k.toFixed(2)} ahead${(rx * ux + ry * uy).toFixed(0)} side${(rx * uy - ry * ux).toFixed(0)}`); }
    b.x += b.vx * DT; b.y += b.vy * DT;
  }
  return { least, vAt };
}
