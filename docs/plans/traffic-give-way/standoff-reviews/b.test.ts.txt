function tourOnLeg(tr: Traffic, fromT: number) {
  for (let T = fromT; T < fromT + 4 * 3600; T += 1) { const vs = tr.vessels(T);
    for (let i = 0; i < vs.length; i++) { const v = vs[i]; if (v.kind === 'tour' && v.moving && Math.abs(v.h - 270) < 1 && Math.abs(v.x - LEG_E.x) < 8 && Math.abs(v.y - LEG_E.y) < 8) return { T, i }; } }
  throw new Error('none');
}
type B = Moving;
function run(i: number, T: number, boats: B[], dur: number, give = true, trace?: (t: number, v: any, g: any, bs: B[], base: any) => void) {
  const tr = new Traffic(true, SEED), tb = new Traffic(true, SEED);
  let least = Infinity, tl = 0, sides: number[] = [], onLand = 0, baseLand = 0, maxOffLand = 0, minK = 1, maxRate = 0, h0 = NaN;
  const bs = boats.map(b => ({ ...b }));
  for (let t = T, n = 0; t < T + dur; t += DT, n++) {
    const v = tr.vessels(t, bs, DT, land, give ? undefined : () => true)[i], hv = vesselHull(v), g = tr.give[i];
    for (const b of bs) { const d = segPt(b, hv.a, hv.b) - hv.r; if (d < least) { least = d; tl = t - T; } }
    if (g.side && sides[sides.length - 1] !== g.side) sides.push(g.side); minK = Math.min(minK, g.k);
    const base = tb.vessels(t)[i];
    if (Math.abs(base.h - (tb as any)._h) < 0.01 && !isNaN(h0)) maxRate = Math.max(maxRate, Math.abs(((v.h - h0 + 540) % 360) - 180) / DT); h0 = v.h; (tb as any)._h = base.h;
    if (n % 30 === 0) { const l = land(v) || land(hv.a) || land(hv.b); if (l) { onLand++; maxOffLand = Math.max(maxOffLand, Math.hypot(g.ox, g.oy)); } if (land(base)) baseLand++; }
    trace?.(t - T, v, g, bs, base);
    for (const b of bs) { b.x += b.vx * DT; b.y += b.vy * DT; }
  }
  return { least: +least.toFixed(1), tl: +tl.toFixed(0), sides: sides.join('>'), minK: +minK.toFixed(2), maxRate: +maxRate.toFixed(1), onLand, baseLand, maxOffLand: +maxOffLand.toFixed(0) };
}
