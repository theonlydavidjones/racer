import { it } from 'vitest';
import { SERVICES, Traffic, VESSEL, vesselHull, type Vessel } from '/home/user/yield/src/sim/traffic';
import { landAt } from '/home/user/yield/src/environment/sea';
import { hitsWall } from '/home/user/yield/src/geo/marina';
import { LL, toLL } from '/home/user/yield/src/geo/geo';
const DT = 1 / 60, R = Math.PI / 180;
const land = (p: any) => !!landAt(p);
const ll = (p: any) => { const q = toLL(p); return `${q.lat.toFixed(4)},${q.lon.toFixed(4)}`; };
it('tour boat westbound off the Wharf, a boat north of her route', () => {
  const find = new Traffic(true, 1); let T = -1; const x0 = LL(37.8128, -122.4160).x;
  for (let t = 0; t < 86400; t += 1) { const v = find.vessels(t)[4]; if (v.moving && Math.abs(v.h - 270) < 2 && Math.abs(v.x - x0) < 6) { T = t; break; } }
  for (const [blat, blon, kt, hdg] of [[37.8134, -122.4200, 0, 0], [37.8136, -122.4215, 0, 0], [37.8134, -122.4235, 3, 90], [37.8140, -122.4230, 4, 110]]) {
    const tr = new Traffic(true, 1), base = new Traffic(true, 1), p = LL(blat, blon), b = { x: p.x, y: p.y, vx: Math.sin(hdg * R) * kt * 0.5144, vy: Math.cos(hdg * R) * kt * 0.5144 };
    let landN = 0, first = '', maxOff = 0, minD = 1e9, stopped = 0;
    for (let i = 0; i < 300 * 60; i++) { const t = T + i * DT, v = tr.vessels(t, [b], DT, land)[4], g = tr.give[4], hv = vesselHull(v), tv = base.vessels(t - g.lag)[4]; b.x += b.vx * DT; b.y += b.vy * DT;
      const on = land(v) || hitsWall(v) || land(hv.a) || land(hv.b) || hitsWall(hv.a) || hitsWall(hv.b);
      if (on && !land(tv)) { landN++; first ||= `${ll(v)} h ${v.h.toFixed(0)} off ${Math.hypot(g.ox, g.oy).toFixed(0)} m at ${(i / 60).toFixed(0)} s`; }
      maxOff = Math.max(maxOff, Math.hypot(g.ox, g.oy)); minD = Math.min(minD, Math.hypot(v.x - b.x, v.y - b.y)); if (g.k < 0.01) stopped++; }
    console.log('boat', blat, blon, kt, 'kt on', hdg, ': hull on land', (landN / 60).toFixed(1), 's', first, 'maxOff', maxOff.toFixed(0), 'minD', minD.toFixed(0), 'stopped', (stopped / 60).toFixed(0), 'lag', tr.give[4].lag.toFixed(0));
  }
});
